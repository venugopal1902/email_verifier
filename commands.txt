# 1. Stop any existing containers (Clean slate)
docker-compose down
docker rm -f $(docker ps -aq)

# 2. Build and Start in the background
Write-Host "Starting Docker Compose..." -ForegroundColor Cyan
docker-compose up --build -d

# 3. Watch the logs to see when it's ready
# (Press Ctrl+C to exit logs once you see "Starting development server...")
docker-compose logs -f web

# 4. Run Database Migrations
Write-Host "Running Migrations..." -ForegroundColor Cyan
docker-compose exec web python manage.py migrate

# 5. (Optional) Create Admin User
docker-compose exec web python manage.py createsuperuser

# 6. Open in Browser
Write-Host "App is running at: http://127.0.0.1:8000/files/dashboard/" -ForegroundColor Green



# 1. Enter the Django Shell
docker-compose exec web python manage.py shell

# 2. Paste this Python code to delete stuck files:
from files.models import FileUpload
deleted_count, _ = FileUpload.objects.filter(status='Processing').delete()
print(f"Deleted {deleted_count} stuck files.")

# 3. Exit the shell
exit()
